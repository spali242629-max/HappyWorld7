function registerUser() { let phone = document.getElementById('phone').value; let pass = document.getElementById('pass').value; if(phone.length < 10 || pass.length < 4) { alert("Sahi details daalo bhai!"); return; } // User ka data ek "dibbe" (Object) mein rakhein let userData = { mobile: phone, password: pass, balance: 500.00 // Naye user ko 500 ka bonus! }; // Phone ki memory (Local Storage) mein save karein localStorage.setItem('user_' + phone, JSON.stringify(userData)); localStorage.setItem('currentUser', phone); // Yaad rakho kaun login hai alert("Registration Successful! ₹500 bonus mila hai."); window.location.href = "home.html"; }
